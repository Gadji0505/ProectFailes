CREATE TABLE PublicationTypes (
  publication_type_id SERIAL PRIMARY KEY,
  type_name VARCHAR(50) NOT NULL UNIQUE
  );
  
  CREATE TABLE Publishers (
  publisher_id SERIAL PRIMARY KEY,
  publisher_name VARCHAR(100) NOT NULL,
  address VARCHAR(255),
  phone VARCHAR(20)
  );
  
  CREATE TABLE Topics (
  topic_id SERIAL PRIMARY KEY,
  topic_name VARCHAR(100) NOT NULL UNIQUE
  );
  
  CREATE TABLE StorageLocations (
  storage_location_id SERIAL PRIMARY KEY,
  location_name VARCHAR(100) NOT NULL UNIQUE,
  description TEXT
  );
  
  CREATE TABLE Books (
  book_id SERIAL PRIMARY KEY,
  title VARCHAR(255) NOT NULL,
  publication_type_id INT REFERENCES PublicationTypes(publication_type_id),
  publisher_id INT REFERENCES Publishers(publisher_id),
  storage_location_id INT REFERENCES StorageLocations(storage_location_id),
  publication_year INT CHECK (publication_year > 0),
  ISBN VARCHAR(20) UNIQUE
  );
  
  CREATE TABLE Subscribers (
  subscriber_id SERIAL PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  membership_date DATE NOT NULL,
  email VARCHAR(100) UNIQUE
  );
  
  CREATE TABLE Librarians (
  librarian_id SERIAL PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  employment_date DATE NOT NULL,
  email VARCHAR(100) UNIQUE
  );
  
  CREATE TABLE Loans (
  loan_id SERIAL PRIMARY KEY,
  book_id INT REFERENCES Books(book_id),
  subscriber_id INT REFERENCES Subscribers(subscriber_id),
  librarian_id INT REFERENCES Librarians(librarian_id),
  loan_date DATE NOT NULL,
  return_date DATE
  );
  
  CREATE TABLE BookAuthors (
      book_author_id SERIAL PRIMARY KEY,
      book_id INT REFERENCES Books(book_id),
      author_name VARCHAR(100) NOT NULL
  );

SELECT 
    b.title,
    COUNT(l.loan_id) AS total_loans,
    ROUND(AVG(l.return_date - l.loan_date), 2) AS avg_loan_days
FROM Books b
LEFT JOIN Loans l ON b.book_id = l.book_id AND l.return_date IS NOT NULL
GROUP BY b.book_id, b.title;

SELECT 
    s.subscriber_id,
    s.name
FROM 
    Subscribers s
WHERE 
    s.subscriber_id NOT IN (
        SELECT DISTINCT l.subscriber_id
        FROM Loans l
        WHERE l.loan_date >= CURRENT_DATE - INTERVAL '6 months'
    );

SELECT 
    b.book_id,
    b.title,
    COUNT(l.loan_id) AS total_loans
FROM Books b
JOIN Loans l ON b.book_id = l.book_id
WHERE l.loan_date <= CURRENT_DATE - INTERVAL '1 year'
GROUP BY b.book_id, b.title
ORDER BY total_loans DESC
LIMIT 3;

SELECT DISTINCT s.subscriber_id, s.name
FROM Subscribers s
JOIN Loans l ON s.subscriber_id = l.subscriber_id
JOIN Books b ON l.book_id = b.book_id
WHERE b.publisher_id = 1
  AND l.loan_date <= CURRENT_DATE - INTERVAL '1 year'
  AND NOT EXISTS (
    SELECT 1
    FROM Loans l2
    JOIN Books b2 ON l2.book_id = b2.book_id
    WHERE l2.subscriber_id = s.subscriber_id
      AND b2.publisher_id <> 1
  );

(SELECT 
    ba.author_name,
    COUNT(l.loan_id) AS total_loans
FROM BookAuthors ba
LEFT JOIN Books b ON ba.book_id = b.book_id
LEFT JOIN Loans l ON b.book_id = l.book_id
WHERE l.loan_date <= CURRENT_DATE - INTERVAL '1 year'
GROUP BY ba.author_name) AS subquery
WHERE 
    total_loans <= 3;
