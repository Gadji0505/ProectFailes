db.Doctor.aggregate([
  {
    $group: {
      _id: "$specialty",
      maxExperience: { $max: "$experience" },
      doctors: { $push: "$$ROOT" }
    }
  },
  {
    $project: {
      doctors: {
        $filter: {
          input: "$doctors",
          as: "doctor",
          cond: { $eq: ["$$doctor.experience", "$maxExperience"] }
        }
      }
    }
  },
  { $unwind: "$doctors" },
  {
    $project: {
      specialty: "$_id",
      name: "$doctors.name",
      experience: "$doctors.experience",
      _id: 0
    }
  }
])
